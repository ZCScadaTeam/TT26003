# 自定义 DLL 使用示例

## 完整示例：创建一个数据处理库

### 1. 创建 DataProcessor.cs

```csharp
using System.Text.Json;

namespace ZC.MyCustomLib.Data;

/// <summary>
/// 数据处理器
/// </summary>
public class DataProcessor
{
    /// <summary>
    /// 将对象序列化为 JSON
    /// </summary>
    public string SerializeToJson<T>(T data)
    {
        return JsonSerializer.Serialize(data, new JsonSerializerOptions 
        { 
            WriteIndented = true 
        });
    }

    /// <summary>
    /// 从 JSON 反序列化对象
    /// </summary>
    public T? DeserializeFromJson<T>(string json)
    {
        return JsonSerializer.Deserialize<T>(json);
    }

    /// <summary>
    /// 验证数据
    /// </summary>
    public bool ValidateData(string data)
    {
        return !string.IsNullOrWhiteSpace(data);
    }
}

/// <summary>
/// 数据模型示例
/// </summary>
public class DataModel
{
    public int Id { get; set; }
    public string Name { get; set; } = string.Empty;
    public DateTime CreatedAt { get; set; }
}
```

### 2. 在 App 项目中使用

#### 步骤 1：添加项目引用

在 `App/App.csproj` 中添加：
```xml
<ItemGroup>
    <ProjectReference Include="../@Core/MyCustomLib/MyCustomLib.csproj"/>
</ItemGroup>
```

#### 步骤 2：在代码中使用

```csharp
using ZC.MyCustomLib;
using ZC.MyCustomLib.Data;

namespace AppZC.Services;

public class MyService
{
    private readonly DataProcessor _processor;

    public MyService()
    {
        _processor = new DataProcessor();
    }

    public void ProcessData()
    {
        // 创建数据
        var data = new DataModel
        {
            Id = 1,
            Name = "测试数据",
            CreatedAt = DateTime.Now
        };

        // 序列化
        string json = _processor.SerializeToJson(data);
        Console.WriteLine(json);

        // 反序列化
        var restored = _processor.DeserializeFromJson<DataModel>(json);
        
        // 验证
        bool isValid = _processor.ValidateData(json);
        
        // 使用辅助类
        string message = MyCustomHelper.FormatMessage($"处理完成: {data.Name}");
        Console.WriteLine(message);
    }
}
```

## 实战示例：创建配置管理库

### 1. 创建 ConfigManager.cs

```csharp
using System.Text.Json;

namespace ZC.MyCustomLib.Config;

/// <summary>
/// 配置管理器
/// </summary>
public class ConfigManager
{
    private readonly string _configPath;

    public ConfigManager(string configPath = "config.json")
    {
        _configPath = configPath;
    }

    /// <summary>
    /// 加载配置
    /// </summary>
    public T? LoadConfig<T>() where T : class
    {
        if (!File.Exists(_configPath))
            return null;

        string json = File.ReadAllText(_configPath);
        return JsonSerializer.Deserialize<T>(json);
    }

    /// <summary>
    /// 保存配置
    /// </summary>
    public void SaveConfig<T>(T config) where T : class
    {
        string json = JsonSerializer.Serialize(config, new JsonSerializerOptions 
        { 
            WriteIndented = true 
        });
        File.WriteAllText(_configPath, json);
    }

    /// <summary>
    /// 配置是否存在
    /// </summary>
    public bool ConfigExists() => File.Exists(_configPath);
}

/// <summary>
/// 应用配置示例
/// </summary>
public class AppConfig
{
    public string AppName { get; set; } = "MyApp";
    public string Version { get; set; } = "1.0.0";
    public DatabaseConfig Database { get; set; } = new();
}

public class DatabaseConfig
{
    public string ConnectionString { get; set; } = string.Empty;
    public int Timeout { get; set; } = 30;
}
```

### 2. 在 App 中使用

```csharp
using ZC.MyCustomLib.Config;

namespace AppZC;

public class Startup
{
    public void Initialize()
    {
        var configManager = new ConfigManager("appsettings.json");

        // 加载配置
        var config = configManager.LoadConfig<AppConfig>();
        
        if (config == null)
        {
            // 创建默认配置
            config = new AppConfig
            {
                AppName = "ZC Application",
                Version = "1.0.0",
                Database = new DatabaseConfig
                {
                    ConnectionString = "Data Source=app.db",
                    Timeout = 30
                }
            };
            
            // 保存配置
            configManager.SaveConfig(config);
        }

        Console.WriteLine($"应用名称: {config.AppName}");
        Console.WriteLine($"版本: {config.Version}");
    }
}
```

## 编译和发布流程

### 1. 开发阶段（使用项目引用）

```bash
# 在 App.csproj 中使用 ProjectReference
dotnet build
```

### 2. 发布阶段（使用 DLL 引用）

```bash
# 编译自定义库
cd @Core/MyCustomLib
dotnet build -c Release

# 复制 DLL 到 libs 目录
copy bin\Release\net10.0\ZC.MyCustomLib.dll ..\..\-\libs\Core\
copy bin\Release\net10.0\ZC.MyCustomLib.pdb ..\..\-\libs\Core\

# 修改 App.csproj 使用 Reference 而不是 ProjectReference
# 然后编译 App
cd ..\..\App
dotnet build -c Release
```

## 调试技巧

### 1. 同时调试多个项目
- 在 Visual Studio 中设置多个启动项目
- 可以同时调试 App 和自定义库

### 2. 使用符号文件 (.pdb)
- 确保 .pdb 文件和 .dll 在同一目录
- 可以在自定义库代码中设置断点
